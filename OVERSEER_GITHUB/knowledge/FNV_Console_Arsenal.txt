# ==================== CORE DEBUG TOGGLES ====================
tgm                  - Toggle God Mode (invincible + unlimited ammo)
tcl                  - Toggle Collision (noclip / fly mode)
tfc                  - Toggle Free Camera (add 1 to freeze time)
tmm 1                - Toggle All Map Markers (1 = reveal all)
tdetect              - Toggle AI Detection
tdw                  - Toggle Debug Window
twf                  - Toggle Wireframe Mode
tt                   - Toggle Trees
tfow                 - Toggle Fog of War (map)
ts                   - Toggle Sky
tll                  - Toggle Light Sources (performance debug)
pcb                  - Purge Cell Buffers (helps with stuttering)

# ==================== PLAYER COMMANDS ====================
player.setlevel [X]               - Set exact player level
player.advlevel                   - Advance one level
player.rewardxp [amount]          - Give XP
player.setav [stat] [value]       - Set attribute/skill (e.g. strength, sneak)
player.modav [stat] [value]       - Modify attribute/skill
player.addperk [perkID]           - Add perk
player.removeperk [perkID]        - Remove perk
player.additem [ID] [amount]      - Add item to player
player.removeitem [ID] [amount]   - Remove item from player
player.setcrimegold 0             - Clear current bounty
player.moveto [refID]             - Teleport player to reference

# ==================== TARGETED COMMANDS ====================
# (Click on NPC/object first)
kill                              - Kill selected target
resurrect                         - Resurrect selected target
disable                           - Disable selected target
enable                            - Enable selected target
inv                               - Show inventory of selected
sexchange                         - Change gender of selected
additem [ID] [amount]             - Add item to selected target
removeitem [ID] [amount]          - Remove item from selected

# ==================== QUEST & STORY ====================
sqv [questID]                     - Show Quest Variables
sqo [questID]                     - Show Quest Objectives
setstage [questID] [stage#]       - Set quest to specific stage
getstage [questID]                - Get current quest stage
completequest [questID]           - Force complete quest
resetquest [questID]              - Reset quest (removes from log)
movetoqt                          - Move to current quest target
showquesttargets                  - List active quest targets

# ==================== POSITION & MOVEMENT ====================
getpos [x/y/z]                    - Get current position axis
setpos [x/y/z] [value]            - Set position
getangle [x/y/z]                  - Get rotation
setangle [x/y/z] [value]          - Set rotation
getscale                          - Get current scale
setscale [value]                  - Set scale (1.0 = normal)

# ==================== PERFORMANCE & ENGINE ====================
setgs fov [value]                 - Set Field of View
setgs fmoverunmult [value]        - Run speed multiplier
setgs fwalkspeedmult [value]      - Walk speed multiplier
setgs fjumpheight [value]         - Jump height
setgs fgamma [value]              - Gamma / brightness
setgs foverencumberedmult [value] - Encumbrance multiplier

# ==================== MISC DEBUG ====================
help                              - List available commands
help "keyword"                    - Search commands containing keyword
getav [stat]                      - Get value of attribute/skill
setav [stat] [value]              - Set value on target
modav [stat] [value]              - Modify value on target
